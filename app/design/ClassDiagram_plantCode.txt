'  UML Relationships:
'  Road --> Car : contains cars in queue
'  JunctionController --> Road : manages multiple roads
'  JunctionController --> JunctionPhase : owns traffic phase logic
'  JunctionPhase --> PhaseValue : uses enum for signal state

@startuml

	+ final int length;
	+ final int crossingTime;
	+ final String plate;
	+ float posInJunction;

	- String generatePlate()
	+ Car (int len, float position, int secToCross)
	+ String toString()
}

	- final int _S;
	- final String _roadName;

}
